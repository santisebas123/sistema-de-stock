Correcciones para gestion.html

Este archivo contiene las correcciones necesarias para eliminar la pantalla blanca y corregir Firebase Auth.

1) Reemplazar onAuthStateChanged

onAuthStateChanged(auth, async user => {

  document.getElementById('loading-screen').classList.add('hidden');

  try {

    if (user) {

      currentUser = user;

      await loadUserData();

      subscribeAll();

      document.getElementById('auth-screen').classList.add('hidden');
      document.getElementById('app-screen').classList.remove('hidden');

      updateSidebar();
      renderDashboard();

    } else {

      currentUser = null;

      unsubs.forEach(u => u());

      document.getElementById('auth-screen').classList.remove('hidden');
      document.getElementById('app-screen').classList.add('hidden');
    }

  } catch (e) {

    console.error('ERROR AUTH:', e);

    alert('Error cargando la aplicación. Revisá la consola.');
  }
});

window.saveProducto = async () => {

  try {

    const nombre = document.getElementById('prod-nombre').value.trim();

    if (!nombre) {
      alert('El nombre es obligatorio');
      return;
    }

    await addDoc(col('productos'), {
      nombre,
      cat: document.getElementById('prod-cat').value || '',
      qty: parseInt(document.getElementById('prod-qty').value) || 0,
      costo: parseInt(document.getElementById('prod-costo').value) || 0,
      precioVenta: parseInt(document.getElementById('prod-precio-venta').value) || 0,
      moneda: document.getElementById('prod-moneda').value,
      emoji: document.getElementById('prod-emoji').value || '📦',
      variantes: state.variants_temp,
      creadoEn: today()
    });

    state.variants_temp = [];

    ['prod-nombre','prod-cat','prod-qty','prod-costo','prod-precio-venta','prod-emoji']
      .forEach(id => document.getElementById(id).value = '');

    renderVariantTags();

    closeModal('modal-producto');

  } catch (e) {

    console.error('ERROR GUARDANDO PRODUCTO:', e);

    alert('No se pudo guardar el producto');
  }
};
